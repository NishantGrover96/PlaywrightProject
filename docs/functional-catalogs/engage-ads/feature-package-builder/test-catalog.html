Eligibility seeded SELECT COUNT(*) FROM EngageAds.package_eligibility WHERE package_seq = :newSeq AND owner_table='PROGRAM_DEALER_TYPE' AND allow_flag=1 AND is_deleted=0 ≥ 10 rows (matches existing packages)
